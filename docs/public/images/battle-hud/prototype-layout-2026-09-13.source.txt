  <script>
    (() => {
      const root = document.getElementById('pb-hud-layout-study');
      const svg = root.querySelector('.pb-timeline svg');
      const ns = 'http://www.w3.org/2000/svg';
      const state = { elapsed: 0, conduct: 'spread' };
      const conductOrder = ['none', 'spread', 'echo'];
      const conductNames = { none: '未選択', spread: 'ひろがり', echo: 'やまびこ' };
      const events = [
        { id: 'a', type: 'arp', fire: 1, notes: [
          { name: 'C♯', tone: 'pink', time: 1, filled: true },
          { name: 'E', tone: 'green', time: 1.65, filled: false },
          { name: 'G♯', tone: 'blue', time: 2.3, filled: true }
        ] },
        { id: 'b', type: 'chord', fire: 3.6, notes: [
          { name: 'C♯', tone: 'pink', time: 3.6, filled: false },
          { name: 'E', tone: 'green', time: 3.6, filled: false },
          { name: 'G♯', tone: 'blue', time: 3.6, filled: false }
        ] }
      ];
      function element(name, attributes, parent, text) {
        const item = document.createElementNS(ns, name);
        Object.entries(attributes || {}).forEach(([key, value]) => item.setAttribute(key, String(value)));
        if (text !== undefined) item.textContent = text;
        (parent || svg).appendChild(item);
        return item;
      }
      function render() {
        root.querySelectorAll('.pb-conduct').forEach(button => {
          const selected = button.dataset.conduct === state.conduct;
          button.classList.toggle('pb-selected', selected);
          button.setAttribute('aria-pressed', String(selected));
        });
        root.querySelector('.pb-conduct-status').textContent = `次回用Conduct：${conductNames[state.conduct]}`;
        const width = Math.max(260, svg.getBoundingClientRect().width || 690);
        const fireX = 16;
        const pps = (width - fireX - 24) / 4.2;
        const noteW = Math.max(24, Math.min(32, .65 * pps - 6));
        const xAt = time => fireX + (time - state.elapsed) * pps;
        const current = events.find(event => event.fire > state.elapsed && !event.notes.every(note => note.filled));
        svg.replaceChildren();
        svg.setAttribute('viewBox', `0 0 ${width} 66`);
        element('title', {}, svg, 'AttackEvent予告の配置と消え方の比較案');
        element('desc', {}, svg, '各音は発射予定時刻で消え、残りの音の時間間隔は詰めない。アルペジオの受付は先頭の発火時刻で終了し、装填対象の強調は次の予告へ移る。');
        const defs = element('defs');
        const clip = element('clipPath', { id: 'pb-hud-study-timeline-clip' }, defs);
        element('rect', { x: 0, y: 0, width, height: 66 }, clip);
        element('line', { x1: fireX, x2: width - 6, y1: 40, y2: 40, stroke: 'var(--pb-border)', 'stroke-width': 1, opacity: .45 });
        element('line', { x1: fireX, x2: fireX, y1: 17, y2: 65, stroke: 'var(--pb-ink)', 'stroke-width': 2 });
        element('text', { x: fireX, y: 12, 'text-anchor': 'middle', fill: 'var(--pb-ink)', 'font-size': 11 }, svg, '発射');
        const layer = element('g', { 'clip-path': 'url(#pb-hud-study-timeline-clip)' });
        events.forEach(event => {
          const remaining = event.notes.filter(note => note.time > state.elapsed);
          if (!remaining.length) return;
          const active = current && current.id === event.id;
          const firstX = xAt(event.fire);
          const lastX = xAt(event.notes[event.notes.length - 1].time);
          const left = Math.max(fireX - 5, firstX - noteW / 2 - 7);
          const right = lastX + noteW / 2 + 7;
          const top = event.type === 'chord' ? 15 : 23;
          const height = event.type === 'chord' ? 49 : 34;
          const group = element('g', { 'aria-label': `${event.type === 'chord' ? 'コード' : 'アルペジオ'}、${active ? '装填対象' : event.fire <= state.elapsed ? '受付終了・発射進行中' : '後続の予告'}` }, layer);
          element('rect', { x: left, y: top, width: Math.max(1, right - left), height, rx: 5, fill: 'var(--pb-panel)', stroke: active ? 'var(--pb-accent)' : 'var(--pb-border)', 'stroke-width': active ? 3 : 1 }, group);
          event.notes.forEach((note, index) => {
            if (note.time <= state.elapsed) return;
            const x = xAt(note.time);
            const cy = event.type === 'chord' ? 24 + index * 16 : 40;
            const nh = event.type === 'chord' ? 14 : 22;
            const tone = `var(--pb-${note.tone})`;
            const mark = element('g', { 'aria-label': `${note.name}、${note.filled ? '装填済み' : '未装填'}` }, group);
            element('rect', { x: x - noteW / 2, y: cy - nh / 2, width: noteW, height: nh, rx: 3, fill: note.filled ? tone : 'var(--pb-panel)', stroke: tone, 'stroke-width': 1.5 }, mark);
            element('text', { x, y: cy + 4, 'text-anchor': 'middle', fill: note.filled ? 'var(--pb-note-ink)' : tone, 'font-size': 11, 'font-weight': 500 }, mark, note.name);
          });
        });
        root.querySelector('.pb-study-status').textContent = state.elapsed < 1 ? 'アルペジオが現在の装填対象です。' : state.elapsed < 2.3 ? 'アルペジオの受付は終了。残りの音が順番に発射時刻を迎え、コードが装填対象です。' : state.elapsed < 3.6 ? 'アルペジオの表示が終了し、コードが装填対象です。' : 'この例のすべての予告が終了しました。';
      }
      render();
      const conductArea = root.querySelector('.pb-conducts');
      function cycleConduct(direction) {
        const index = conductOrder.indexOf(state.conduct);
        state.conduct = conductOrder[(index + direction + conductOrder.length) % conductOrder.length];
        render();
      }
      root.querySelectorAll('.pb-conduct').forEach(button => {
        button.addEventListener('click', () => { state.conduct = button.dataset.conduct; render(); });
      });
      let wheelTotal = 0;
      let lastWheelAt = 0;
      let lastWheelDirection = 0;
      conductArea.addEventListener('wheel', event => {
        if (event.ctrlKey || !event.deltaY) return;
        event.preventDefault();
        const now = performance.now();
        const direction = Math.sign(event.deltaY);
        if (now - lastWheelAt > 200 || direction !== lastWheelDirection) wheelTotal = 0;
        lastWheelAt = now;
        lastWheelDirection = direction;
        const scale = event.deltaMode === 1 ? 16 : event.deltaMode === 2 ? 100 : 1;
        wheelTotal += event.deltaY * scale;
        if (Math.abs(wheelTotal) >= 40) { cycleConduct(direction); wheelTotal = 0; }
      }, { passive: false });
      conductArea.addEventListener('keydown', event => {
        if (event.key !== 'ArrowDown' && event.key !== 'ArrowUp') return;
        event.preventDefault();
        cycleConduct(event.key === 'ArrowDown' ? 1 : -1);
      });
      const observer = new ResizeObserver(render);
      observer.observe(svg);
      if (globalThis.Tweak) {
        const tweak = new Tweak({ container: root, onChange: render });
        tweak.addSlider(state, 'elapsed', { label: '経過時間（譜面の例）', min: 0, max: 3.7, step: .05, unit: '秒' });
        tweak.addSelect(state, 'conduct', { label: '次回用Conduct', options: [{ label: '未選択', value: 'none' }, { label: 'ひろがり', value: 'spread' }, { label: 'やまびこ', value: 'echo' }] });
      }
    })();
  </script>
</div>
