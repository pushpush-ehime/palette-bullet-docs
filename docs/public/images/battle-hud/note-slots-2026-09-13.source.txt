  <script>
    (() => {
      const root = document.getElementById('pb-preview-note-study');
      const state = { loading: 'partial', shape: 'angular' };
      const noteNames = ['Cシャープ', 'E', 'Gシャープ'];
      function render() {
        root.style.setProperty('--pb-radius', state.shape === 'round' ? '26px' : '9px');
        root.querySelectorAll('.pb-demo-note').forEach(note => {
          const index = Number(note.dataset.entry);
          const filled = state.loading === 'full' || (state.loading === 'partial' && index !== 1);
          note.dataset.filled = String(filled);
          const prefix = note.closest('.pb-arp-chain') ? `${index + 1}番目、` : '';
          note.setAttribute('aria-label', `${prefix}${noteNames[index]}、${filled ? '装填済み' : '未装填'}`);
        });
        const caption = state.loading === 'empty' ? 'すべて未装填' : state.loading === 'full' ? 'すべて装填済み' : 'C♯・G♯は装填済み／Eが不足';
        root.querySelectorAll('.pb-state-caption').forEach(label => { label.textContent = caption; });
      }
      render();
      if (globalThis.Tweak) {
        const tweak = new Tweak({ container: root, onChange: render });
        tweak.addSelect(state, 'loading', { label: '装填状態', options: [{ label: 'すべて未装填', value: 'empty' }, { label: '一部装填', value: 'partial' }, { label: 'すべて装填済み', value: 'full' }] });
        tweak.addSelect(state, 'shape', { label: '音枠の形', options: [{ label: '角形', value: 'angular' }, { label: '丸形', value: 'round' }] });
      }
    })();
  </script>
